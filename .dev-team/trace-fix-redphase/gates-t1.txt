# Gate 2b 脚本全文（保存为任意路径后 python3 <script> crates/core/src/trace_v2.rs 即可重放）：
  import re, sys
  p = sys.argv[1]
  src = open(p, encoding="utf-8").read()
  decls = re.findall(r'pub fn\s+([A-Za-z0-9_]+)', src)
  body_re = re.compile(r'pub fn\s+([A-Za-z0-9_]+)\s*\([^)]*\)\s*->\s*[^{]+\{(.*?)\n\}', re.S)
  bodies = {m.group(1): m.group(2) for m in body_re.finditer(src)}
  errs = []
  if len(decls) != len(bodies):
      errs.append(f"函数体解析不完整: decls={len(decls)} parsed={len(bodies)}")
  for name, body in bodies.items():
      stmts = [l.strip() for l in body.splitlines() if l.strip() and not l.strip().startswith("//")]
      if stmts != ["todo!()"]:
          errs.append(f"{name}: 函数体语句必须恰为 todo!()，实得 {stmts}")
      if len(stmts) > 3:
          errs.append(f"{name}: 有效行数 {len(stmts)} > 3")
      for kw in ("let ", "if ", "match ", "for ", "while ", "return ", "?"):
          if kw in body:
              errs.append(f"{name}: 空桩体内出现逻辑关键字 `{kw}`")
  TURN_DIAG = ["pub error_contract: Option<String>", "pub status_conflict: bool", "pub error_missing: bool"]
  MERGED = ["pub turn_id: String", "pub seq: i64", "pub status: String", "pub payload: Value"]
  for struct, fields in (("TurnDiagnostics", TURN_DIAG), ("MergedTurn", MERGED)):
      m = re.search(rf'pub struct {struct}\s*\{{(.*?)\n\}}', src, re.S)
      if not m:
          errs.append(f"缺少 `pub struct {struct}`")
          continue
      for f in fields:
          if f not in m.group(1):
              errs.append(f"{struct} 缺少字段 `{f}`")
  required_fns = ["annotate_turn", "merge_turn", "aggregate_session", "turn_status_of", "is_terminal_status"]
  for fn in required_fns:
      if fn not in bodies:
          errs.append(f"缺少签名 `pub fn {fn}`")
  if errs:
      print(f"[FAIL] rust-stub-purity: {len(errs)} 处违规")
      for e in errs:
          print(f"  - {e}")
      sys.exit(1)
  print(f"[PASS] rust-stub-purity: {len(bodies)} 个函数体均为单条 todo!()，2 个结构体字段与 5 个签名齐备")
